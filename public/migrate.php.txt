/**
 * Web Migration Runner - Access restricted to Super Admin only
 *
 * @package EduTrack
 * @filepath public/migrate.php
 * @version 2.0
 */

$projectRoot = dirname(__DIR__);
require_once $projectRoot . '/config/config.php';

session_start();

// Restrict to Super Admin only
if (!isset($_SESSION['logged_in']) || $_SESSION['logged_in'] !== true) {
    die('Access denied. Please login as Super Admin.');
}

$isSuperAdmin = $_SESSION['is_super_admin'] ?? false;
if (!$isSuperAdmin) {
    die('Access denied. Super Admin privileges required.');
}

require_once $projectRoot . '/migrations/migration.php';

$migration = new DatabaseMigration();
$message = '';
$status = '';

// Handle actions
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (isset($_POST['action'])) {
        switch ($_POST['action']) {
            case 'migrate':
                $result = $migration->migrate();
                $message = $result['message'];
                $status = $result['success'] ? 'success' : 'danger';
                break;
            case 'rollback':
                $result = $migration->rollback();
                $message = $result['message'];
                $status = $result['success'] ? 'success' : 'danger';
                break;
            case 'create':
                if (!empty($_POST['description'])) {
                    $result = $migration->createMigration($_POST['description']);
                    $message = $result['success'] ? 'Migration created: ' . $result['file'] : $result['message'];
                    $status = $result['success'] ? 'success' : 'danger';
                }
                break;
        }
    }
}

$executed = $migration->getExecutedMigrations();
$pending = $migration->getPendingMigrations();
$allFiles = $migration->getMigrationFiles();
